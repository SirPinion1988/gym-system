-- ==========================================================
-- 1. TABLA DE USUARIOS DEL SISTEMA (ADMIN Y OPERADORES)
-- ==========================================================
CREATE TABLE IF NOT EXISTS usuarios_sistema (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    rol VARCHAR(20) DEFAULT 'OPERADOR',
    activo BOOLEAN DEFAULT TRUE
);

ALTER TABLE usuarios_sistema ADD COLUMN IF NOT EXISTS rol VARCHAR(20) DEFAULT 'OPERADOR';
ALTER TABLE usuarios_sistema ADD COLUMN IF NOT EXISTS activo BOOLEAN DEFAULT TRUE;

-- Garantizar permisos totales al usuario administrador inicial
UPDATE usuarios_sistema SET rol = 'ADMIN', activo = TRUE WHERE username = 'admin';

-- ==========================================================
-- 2. TABLA DE PLANES Y TARIFAS
-- ==========================================================
CREATE TABLE IF NOT EXISTS planes (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    precio FLOAT NOT NULL,
    dias_duracion INTEGER DEFAULT 30,
    descripcion VARCHAR(200),
    activo BOOLEAN DEFAULT TRUE
);

ALTER TABLE planes ADD COLUMN IF NOT EXISTS dias_duracion INTEGER DEFAULT 30;
ALTER TABLE planes ADD COLUMN IF NOT EXISTS activo BOOLEAN DEFAULT TRUE;
UPDATE planes SET activo = TRUE WHERE activo IS NULL;

-- Planes iniciales por defecto si no existen
INSERT INTO planes (nombre, precio, dias_duracion, descripcion, activo)
SELECT 'Pase Libre Mensual', 25000.0, 30, 'Acceso libre e ilimitado a musculación y clases.', TRUE
WHERE NOT EXISTS (SELECT 1 FROM planes WHERE nombre = 'Pase Libre Mensual');

INSERT INTO planes (nombre, precio, dias_duracion, descripcion, activo)
SELECT '3 Veces por Semana', 18000.0, 30, 'Hasta 3 ingresos semanales.', TRUE
WHERE NOT EXISTS (SELECT 1 FROM planes WHERE nombre = '3 Veces por Semana');

-- ==========================================================
-- 3. TABLA DE PROFESORES Y ENTRENADORES (CON SUELDOS)
-- ==========================================================
CREATE TABLE IF NOT EXISTS profesores (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    dni VARCHAR(20) UNIQUE NOT NULL,
    celular VARCHAR(30) NOT NULL,
    especialidad VARCHAR(100),
    sueldo FLOAT DEFAULT 0.0,
    tipo_sueldo VARCHAR(30) DEFAULT 'MENSUAL',
    activo BOOLEAN DEFAULT TRUE
);

ALTER TABLE profesores ADD COLUMN IF NOT EXISTS sueldo FLOAT DEFAULT 0.0;
ALTER TABLE profesores ADD COLUMN IF NOT EXISTS tipo_sueldo VARCHAR(30) DEFAULT 'MENSUAL';
ALTER TABLE profesores ADD COLUMN IF NOT EXISTS activo BOOLEAN DEFAULT TRUE;
UPDATE profesores SET sueldo = 0.0 WHERE sueldo IS NULL;
UPDATE profesores SET tipo_sueldo = 'MENSUAL' WHERE tipo_sueldo IS NULL;
UPDATE profesores SET activo = TRUE WHERE activo IS NULL;

-- TABLA DE PAGOS REALES Y LIQUIDACIONES A PROFESORES
CREATE TABLE IF NOT EXISTS pagos_profesores (
    id SERIAL PRIMARY KEY,
    profesor_id INTEGER NOT NULL REFERENCES profesores(id) ON DELETE CASCADE,
    monto FLOAT NOT NULL,
    concepto VARCHAR(150) DEFAULT 'Liquidación de Sueldo / Honorarios',
    metodo_pago VARCHAR(50) DEFAULT 'EFECTIVO',
    fecha_pago TIMESTAMP DEFAULT NOW()
);

-- ==========================================================
-- 4. TABLA DE ACTIVIDADES / CLASES
-- ==========================================================
CREATE TABLE IF NOT EXISTS actividades (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    dias VARCHAR(100) NOT NULL,
    horario VARCHAR(50) NOT NULL,
    cupo_maximo INTEGER DEFAULT 20,
    profesor_id INTEGER REFERENCES profesores(id) ON DELETE SET NULL
);

-- ==========================================================
-- 5. TABLA DE SOCIOS (CON FOTO, APTO MÉDICO Y PRÓRROGAS)
-- ==========================================================
CREATE TABLE IF NOT EXISTS socios (
    id SERIAL PRIMARY KEY,
    dni VARCHAR(20) UNIQUE NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    fecha_nacimiento DATE NOT NULL,
    edad INTEGER NOT NULL,
    celular VARCHAR(30) NOT NULL,
    email VARCHAR(120) UNIQUE NOT NULL,
    foto_base64 TEXT,
    apto_medico_realizacion DATE,
    apto_medico_vencimiento DATE,
    qr_token VARCHAR(100) UNIQUE NOT NULL,
    estado_cuota VARCHAR(20) DEFAULT 'ACTIVO',
    fecha_vencimiento_cuota DATE,
    plan_id INTEGER REFERENCES planes(id) ON DELETE SET NULL,
    bloqueado_manual BOOLEAN DEFAULT FALSE,
    habilitacion_manual_hasta DATE,
    tarjeta_tokenizada BOOLEAN DEFAULT FALSE,
    tarjeta_marca VARCHAR(30),
    tarjeta_ultimos4 VARCHAR(4),
    creado_en TIMESTAMP DEFAULT NOW()
);

ALTER TABLE socios ADD COLUMN IF NOT EXISTS plan_id INTEGER REFERENCES planes(id) ON DELETE SET NULL;
ALTER TABLE socios ADD COLUMN IF NOT EXISTS foto_base64 TEXT;
ALTER TABLE socios ADD COLUMN IF NOT EXISTS apto_medico_realizacion DATE;
ALTER TABLE socios ADD COLUMN IF NOT EXISTS apto_medico_vencimiento DATE;
ALTER TABLE socios ADD COLUMN IF NOT EXISTS bloqueado_manual BOOLEAN DEFAULT FALSE;
ALTER TABLE socios ADD COLUMN IF NOT EXISTS habilitacion_manual_hasta DATE;
ALTER TABLE socios ADD COLUMN IF NOT EXISTS tarjeta_tokenizada BOOLEAN DEFAULT FALSE;
ALTER TABLE socios ADD COLUMN IF NOT EXISTS tarjeta_marca VARCHAR(30);
ALTER TABLE socios ADD COLUMN IF NOT EXISTS tarjeta_ultimos4 VARCHAR(4);
ALTER TABLE socios ADD COLUMN IF NOT EXISTS creado_en TIMESTAMP DEFAULT NOW();

-- ==========================================================
-- 6. TABLA DE EVOLUCIÓN FÍSICA Y MÉTRICAS CORPORALES
-- ==========================================================
CREATE TABLE IF NOT EXISTS evolucion_socio (
    id SERIAL PRIMARY KEY,
    socio_id INTEGER NOT NULL REFERENCES socios(id) ON DELETE CASCADE,
    fecha DATE DEFAULT CURRENT_DATE,
    peso_kg FLOAT,
    altura_cm FLOAT,
    porcentaje_grasa FLOAT,
    cintura_cm FLOAT,
    pecho_cm FLOAT,
    brazo_cm FLOAT,
    cadera_cm FLOAT,
    notas TEXT
);

-- ==========================================================
-- 7. TABLAS DE RUTINAS Y ENTRENAMIENTOS (100% INCLUIDAS)
-- ==========================================================
CREATE TABLE IF NOT EXISTS rutinas (
    id SERIAL PRIMARY KEY,
    socio_id INTEGER NOT NULL REFERENCES socios(id) ON DELETE CASCADE,
    profesor_id INTEGER REFERENCES profesores(id) ON DELETE SET NULL,
    nombre_rutina VARCHAR(120) NOT NULL,
    objetivo VARCHAR(150),
    activa BOOLEAN DEFAULT TRUE,
    creado_en TIMESTAMP DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS ejercicios_rutina (
    id SERIAL PRIMARY KEY,
    rutina_id INTEGER NOT NULL REFERENCES rutinas(id) ON DELETE CASCADE,
    dia_grupo VARCHAR(50) NOT NULL,
    ejercicio VARCHAR(120) NOT NULL,
    series INTEGER DEFAULT 4,
    repeticiones VARCHAR(30) DEFAULT '10-12',
    peso_sugerido VARCHAR(30) DEFAULT '',
    descanso VARCHAR(30) DEFAULT '60s',
    completado BOOLEAN DEFAULT FALSE
);

-- ==========================================================
-- 8. TABLA DE COBROS Y PAGOS (CUOTAS)
-- ==========================================================
CREATE TABLE IF NOT EXISTS pagos (
    id SERIAL PRIMARY KEY,
    socio_id INTEGER NOT NULL REFERENCES socios(id) ON DELETE CASCADE,
    monto FLOAT NOT NULL,
    metodo_pago VARCHAR(50) DEFAULT 'EFECTIVO',
    concepto VARCHAR(150) DEFAULT 'Pago de Cuota',
    external_payment_id VARCHAR(100) UNIQUE,
    fecha_pago TIMESTAMP DEFAULT NOW()
);

ALTER TABLE pagos ADD COLUMN IF NOT EXISTS external_payment_id VARCHAR(100) UNIQUE;

-- ==========================================================
-- 9. TABLA DE FACTURAS ELECTRÓNICAS (ARCA / ex AFIP)
-- ==========================================================
CREATE TABLE IF NOT EXISTS facturas (
    id SERIAL PRIMARY KEY,
    pago_id INTEGER NOT NULL REFERENCES pagos(id) ON DELETE CASCADE,
    socio_id INTEGER NOT NULL REFERENCES socios(id) ON DELETE CASCADE,
    tipo_comprobante VARCHAR(10) DEFAULT 'C',
    punto_venta INTEGER DEFAULT 1,
    numero_comprobante INTEGER NOT NULL,
    cae VARCHAR(50) NOT NULL,
    cae_vencimiento DATE NOT NULL,
    monto_total FLOAT NOT NULL,
    fecha_emision TIMESTAMP DEFAULT NOW(),
    enviada_por_mail BOOLEAN DEFAULT FALSE
);

-- Forzar relación en cascada para evitar errores de integridad al anular cobros
ALTER TABLE facturas DROP CONSTRAINT IF EXISTS facturas_pago_id_fkey;
ALTER TABLE facturas ADD CONSTRAINT facturas_pago_id_fkey FOREIGN KEY (pago_id) REFERENCES pagos(id) ON DELETE CASCADE;

-- ==========================================================
-- 10. TABLA DE PRODUCTOS / KIOSCO (PUNTO DE VENTA)
-- ==========================================================
CREATE TABLE IF NOT EXISTS productos (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    categoria VARCHAR(50) DEFAULT 'BEBIDAS',
    precio_venta FLOAT NOT NULL,
    stock INTEGER DEFAULT 0,
    activo BOOLEAN DEFAULT TRUE
);

-- ==========================================================
-- 11. TABLA DE VENTAS DEL KIOSCO
-- ==========================================================
CREATE TABLE IF NOT EXISTS ventas_productos (
    id SERIAL PRIMARY KEY,
    producto_id INTEGER NOT NULL REFERENCES productos(id) ON DELETE CASCADE,
    cantidad INTEGER DEFAULT 1,
    total FLOAT NOT NULL,
    metodo_pago VARCHAR(50) DEFAULT 'EFECTIVO',
    fecha TIMESTAMP DEFAULT NOW()
);

-- ==========================================================
-- 12. TABLA DE AUDITORÍA Y CONTROL DE ACCESOS (MOLINETE)
-- ==========================================================
CREATE TABLE IF NOT EXISTS registros_acceso (
    id SERIAL PRIMARY KEY,
    socio_id INTEGER NOT NULL REFERENCES socios(id) ON DELETE CASCADE,
    fecha_hora TIMESTAMP DEFAULT NOW(),
    resultado VARCHAR(20) NOT NULL,
    motivo VARCHAR(100) NOT NULL
);

-- ==========================================================
-- 13. ÍNDICES DE RENDIMIENTO Y CONSULTA RÁPIDA
-- ==========================================================
CREATE INDEX IF NOT EXISTS idx_usuarios_username ON usuarios_sistema(username);
CREATE INDEX IF NOT EXISTS idx_socios_dni ON socios(dni);
CREATE INDEX IF NOT EXISTS idx_socios_email ON socios(email);
CREATE INDEX IF NOT EXISTS idx_socios_qr_token ON socios(qr_token);
CREATE INDEX IF NOT EXISTS idx_rutinas_socio ON rutinas(socio_id);
CREATE INDEX IF NOT EXISTS idx_ejercicios_rutina ON ejercicios_rutina(rutina_id);
CREATE INDEX IF NOT EXISTS idx_evolucion_socio ON evolucion_socio(socio_id);
CREATE INDEX IF NOT EXISTS idx_pagos_fecha ON pagos(fecha_pago);
CREATE INDEX IF NOT EXISTS idx_pagos_ext_id ON pagos(external_payment_id);
CREATE INDEX IF NOT EXISTS idx_pagos_profes ON pagos_profesores(profesor_id);
CREATE INDEX IF NOT EXISTS idx_facturas_cae ON facturas(cae);
CREATE INDEX IF NOT EXISTS idx_ventas_fecha ON ventas_productos(fecha);
CREATE INDEX IF NOT EXISTS idx_accesos_fecha ON registros_acceso(fecha_hora);
